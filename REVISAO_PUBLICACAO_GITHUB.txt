+## Registrar uma partida
+
+```python
+from src.db import connect, init_db
+from src.partidas import registrar_partida
+
+conn = connect("cs2.sqlite")
+try:
+    init_db(conn)
+    with conn:
+        partida_id = registrar_partida(
+            conn, "2026-10-20", "de_mirage", "vitoria", 13, 7
+        )
+    print(partida_id)
+finally:
+    conn.close()
+```
+
+Os cinco valores correspondem, na ordem, às colunas `data_partida`, `mapa`,
+`resultado`, `rounds_time` e `rounds_adversario`. O ID retornado é gerado pelo
+SQLite. A função usa SQL parametrizado e deixa a validação para o schema;
+violações de integridade geram `sqlite3.IntegrityError`.
+
+A função não faz `commit` nem fecha a conexão. No exemplo, `with conn` confirma
+a transação ao terminar com sucesso ou desfaz as alterações se ocorrer uma
+exceção. Também é possível usar `conn.commit()` e `conn.rollback()` explicitamente.
+Receber um ID não significa que a transação já foi confirmada.
+
+## Registrar uma performance
+
+Use uma conexão criada por `connect()` e IDs de jogador e partida existentes.

+"""Registro de partidas com as regras de integridade do SQLite."""
+
+import sqlite3
+
+
+def registrar_partida(
+    conn: sqlite3.Connection,
+    data_partida: str,
+    mapa: str,
+    resultado: str,
+    rounds_time: int,
+    rounds_adversario: int,
+) -> int:
+    """Insere uma partida e retorna seu ID, sem confirmar a transação.
+
+    A data deve seguir AAAA-MM-DD; o resultado deve ser 'vitoria',
+    'derrota' ou 'empate', coerente com o placar. O schema valida os
+    valores e os erros do SQLite são propagados para quem chamou.
+    Quem fornece a conexão controla commit, rollback e fechamento.
+    """
+    cursor = conn.execute(
+        "INSERT INTO partida "
+        "(data_partida, mapa, resultado, rounds_time, rounds_adversario) "
+        "VALUES (?, ?, ?, ?, ?)",
+        (data_partida, mapa, resultado, rounds_time, rounds_adversario),
+    )
+    return cursor.lastrowid

+"""Registro dos dados brutos de desempenho de um jogador em uma partida."""

+import sqlite3

+
+def registrar_performance(
+    conn: sqlite3.Connection,
+    *,
+    player_id: int,
+    partida_id: int,
+    kills: int,
+    deaths: int,
+    assists: int,
+    headshots: int,
+    damage: int,
+    utility_damage: int,
+    enemies_blinded: int,
+    flash_assists: int,
+    rounds_played: int,
+    first_kills: int,
+    first_deaths: int,
+    kast_rounds: int,
+) -> int:
+    """Insere uma performance e retorna seu ID, sem confirmar a transação.
+
+    Os IDs devem identificar jogador e partida existentes. Os demais
+    parâmetros correspondem às colunas de dados brutos do schema.
+    O SQLite valida os valores, os relacionamentos e a duplicidade;
+    suas exceções são propagadas. Quem fornece a conexão controla
+    commit, rollback e fechamento.
+    """
+    cursor = conn.execute(
+        "INSERT INTO performance ("
+        "player_id, partida_id, kills, deaths, assists, headshots, damage, "
+        "utility_damage, enemies_blinded, flash_assists, rounds_played, "
+        "first_kills, first_deaths, kast_rounds"
+        ") VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
+        (player_id, partida_id, kills, deaths, assists, headshots, damage,
+         utility_damage, enemies_blinded, flash_assists, rounds_played,
+         first_kills, first_deaths, kast_rounds),
+    )
+    return cursor.lastrowid
